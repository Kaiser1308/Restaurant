```csharp
using Restaurant.Api.Common.Exceptions;
using Restaurant.Api.Domain.Entities;
using Restaurant.Api.Domain.Enums;
using Restaurant.Api.DTOs.Orders;
using Restaurant.Api.Infrastructure.Auth;
using Restaurant.Api.Repositories;

namespace Restaurant.Api.Tests.Services;

public sealed class OrderServiceTests
{
    private readonly Mock<IOrderRepository> _orderRepo = new();
    private readonly Mock<ITableRepository> _tableRepo = new();
    private readonly Mock<IMenuItemRepository> _menuItemRepo = new();
    private readonly TenantContext _tenantContext = new();

    public OrderServiceTests()
    {
        _tenantContext.Set(Guid.NewGuid(), Guid.NewGuid(), UserRole.Waiter, true);
    }

    private OrderService CreateService() => new(_orderRepo.Object, _tableRepo.Object, _menuItemRepo.Object, _tenantContext);

    [Fact]
    public async Task CreateAsync_DuplicateActiveOrder_ThrowsConflictException()
    {
        var tableId = Guid.NewGuid();
        _tableRepo.Setup(r => r.GetByIdAsync(tableId, default))
            .ReturnsAsync(new RestaurantTable { Id = tableId });

        _orderRepo.Setup(r => r.GetActiveOrderByTableAsync(tableId, default))
            .ReturnsAsync(new Order { Id = Guid.NewGuid() });

        var service = CreateService();
        var act = () => service.CreateAsync(new CreateOrderRequest(tableId));

        await act.Should().ThrowAsync<ConflictException>()
            .WithMessage("Table already has an active order.");
    }

    [Fact]
    public async Task CancelItemAsync_SentItem_ThrowsConflictException()
    {
        var itemId = Guid.NewGuid();
        _orderRepo.Setup(r => r.GetItemByIdAsync(itemId, default))
            .ReturnsAsync(new OrderItem
            {
                Id = itemId,
                Status = OrderItemStatus.SentToKitchen,
                OrderId = Guid.NewGuid()
            });

        var service = CreateService();
        var act = () => service.CancelItemAsync(itemId, new CancelOrderItemRequest("some reason"));

        await act.Should().ThrowAsync<ConflictException>()
            .WithMessage("Only pending item can be cancelled.");
    }

    [Fact]
    public async Task AddItemAsync_OrderNotEditable_ThrowsBusinessException()
    {
        var orderId = Guid.NewGuid();
        _orderRepo.Setup(r => r.GetByIdAsync(orderId, default))
            .ReturnsAsync(new Order { Id = orderId, Status = OrderStatus.Paid });

        var service = CreateService();
        var act = () => service.AddItemAsync(orderId, new AddOrderItemRequest(Guid.NewGuid(), 1));

        await act.Should().ThrowAsync<BusinessException>()
            .WithMessage("Order cannot be edited.");
    }
}
```

```csharp
using Microsoft.EntityFrameworkCore.Storage;
using Restaurant.Api.Common.Exceptions;
using Restaurant.Api.Domain.Entities;
using Restaurant.Api.Domain.Enums;
using Restaurant.Api.DTOs.Bills;
using Restaurant.Api.Infrastructure.Auth;
using Restaurant.Api.Repositories;
using Restaurant.Api.Services;

namespace Restaurant.Api.Tests.Services;

public sealed class BillServiceTests
{
    private readonly Mock<IBillRepository> _billRepo = new();
    private readonly Mock<IAuditLogRepository> _auditLogRepo = new();
    private readonly Mock<IPermissionService> _permissionService = new();
    private readonly TenantContext _tenantContext = new();

    public BillServiceTests()
    {
        _tenantContext.Set(Guid.NewGuid(), Guid.NewGuid(), UserRole.Cashier, true);
    }

    private BillService CreateService() => new(_billRepo.Object, _auditLogRepo.Object, _permissionService.Object, _tenantContext);

    private Order CreateOrderWithItems(Guid orderId)
    {
        var tableId = Guid.NewGuid();
        return new Order
        {
            Id = orderId,
            TableId = tableId,
            Status = OrderStatus.SentToKitchen,
            Items = new List<OrderItem>
            {
                new()
                {
                    Id = Guid.NewGuid(),
                    MenuItemId = Guid.NewGuid(),
                    ItemNameSnapshot = "Pho Bo",
                    Quantity = 2,
                    UnitPrice = 50000m,
                    Status = OrderItemStatus.SentToKitchen
                },
                new()
                {
                    Id = Guid.NewGuid(),
                    MenuItemId = Guid.NewGuid(),
                    ItemNameSnapshot = "Tra Da",
                    Quantity = 1,
                    UnitPrice = 10000m,
                    Status = OrderItemStatus.SentToKitchen
                }
            },
            Table = new RestaurantTable { Id = tableId, Name = "Table 1", Status = TableStatus.Occupied }
        };
    }

    [Fact]
    public async Task PayOrderAsync_CreatesBillItemSnapshots()
    {
        var orderId = Guid.NewGuid();
        var order = CreateOrderWithItems(orderId);
        var tenantId = _tenantContext.RequireTenantId();

        _billRepo.Setup(r => r.BeginTransactionAsync(default))
            .ReturnsAsync(Mock.Of<IDbContextTransaction>());
        _billRepo.Setup(r => r.GetOrderForPaymentAsync(orderId, default))
            .ReturnsAsync(order);
        _billRepo.Setup(r => r.NextBillNumberAsync(tenantId, It.IsAny<DateOnly>(), default))
            .ReturnsAsync("20260512-0001");

        var service = CreateService();
        var result = await service.PayOrderAsync(orderId, new PayOrderRequest("Cash"));

        result.Should().NotBeNull();
        result.TotalAmount.Should().Be(110000m); // 2*50000 + 1*10000

        _billRepo.Verify(r => r.AddBillAsync(It.Is<Bill>(b =>
            b.Items.Count == 2 &&
            b.Items[0].ItemNameSnapshot == "Pho Bo" &&
            b.Items[0].UnitPriceSnapshot == 50000m &&
            b.Items[0].Quantity == 2 &&
            b.Items[0].LineTotal == 100000m &&
            b.Items[1].ItemNameSnapshot == "Tra Da" &&
            b.Items[1].UnitPriceSnapshot == 10000m &&
            b.Items[1].Quantity == 1 &&
            b.Items[1].LineTotal == 10000m
        ), default), Times.Once);
    }

    [Fact]
    public async Task VoidBillAsync_NonOwner_ThrowsForbiddenException()
    {
        _tenantContext.Set(_tenantContext.UserId!.Value, _tenantContext.TenantId!.Value, UserRole.Cashier, true);
        _permissionService.Setup(p => p.CanVoidBill(UserRole.Cashier)).Returns(false);

        var service = CreateService();
        var act = () => service.VoidAsync(Guid.NewGuid(), new VoidBillRequest("wrong order"));

        await act.Should().ThrowAsync<ForbiddenException>()
            .WithMessage("Only owner can void bills.");
    }

    [Fact]
    public async Task PayOrderAsync_BillNumberSequence_UsesNextBillNumber()
    {
        var orderId = Guid.NewGuid();
        var order = CreateOrderWithItems(orderId);
        var tenantId = _tenantContext.RequireTenantId();

        _billRepo.Setup(r => r.BeginTransactionAsync(default))
            .ReturnsAsync(Mock.Of<IDbContextTransaction>());
        _billRepo.Setup(r => r.GetOrderForPaymentAsync(orderId, default))
            .ReturnsAsync(order);
        _billRepo.Setup(r => r.NextBillNumberAsync(tenantId, It.IsAny<DateOnly>(), default))
            .ReturnsAsync("20260512-0042");

        var service = CreateService();
        var result = await service.PayOrderAsync(orderId, new PayOrderRequest("Cash"));

        result.BillNumber.Should().Be("20260512-0042");
    }
}
```

```csharp
using Microsoft.AspNetCore.Identity;
using Microsoft.Extensions.Options;
using Restaurant.Api.Domain.Entities;
using Restaurant.Api.Domain.Enums;
using Restaurant.Api.DTOs.Auth;
using Restaurant.Api.Infrastructure.Auth;
using Restaurant.Api.Repositories;

namespace Restaurant.Api.Tests.Services;

public sealed class AuthServiceTests
{
    private readonly Mock<IAuthRepository> _authRepo = new();
    private readonly Mock<IJwtTokenService> _jwtService = new();
    private readonly IOptions<JwtOptions> _jwtOptions = Options.Create(new JwtOptions { RefreshTokenDays = 7 });

    private readonly User _testUser = new()
    {
        Id = Guid.NewGuid(),
        TenantId = Guid.NewGuid(),
        Username = "owner",
        Name = "Owner",
        Role = UserRole.Owner,
        IsActive = true
    };

    private AuthService CreateService() => new(_authRepo.Object, _jwtService.Object, _jwtOptions);

    [Fact]
    public async Task RefreshAsync_ValidToken_RotatesRefreshToken()
    {
        var oldTokenHash = "old-hash";
        var newRefreshToken = "new-plain-token";
        var newTokenHash = "new-hash";

        var storedToken = new RefreshToken
        {
            Id = Guid.NewGuid(),
            UserId = _testUser.Id,
            TokenHash = oldTokenHash,
            ExpiresAt = DateTimeOffset.UtcNow.AddDays(1),
            User = _testUser
        };

        _jwtService.Setup(j => j.HashRefreshToken("old-token")).Returns(oldTokenHash);
        _authRepo.Setup(r => r.GetRefreshTokenByHashAsync(oldTokenHash, default))
            .ReturnsAsync(storedToken);
        _jwtService.Setup(j => j.CreateAccessToken(_testUser)).Returns("access-token");
        _jwtService.Setup(j => j.CreateRefreshToken()).Returns(newRefreshToken);
        _jwtService.Setup(j => j.HashRefreshToken(newRefreshToken)).Returns(newTokenHash);

        var service = CreateService();
        var result = await service.RefreshAsync(new RefreshTokenRequest("old-token"));

        result.Should().NotBeNull();
        result.AccessToken.Should().Be("access-token");
        result.RefreshToken.Should().Be(newRefreshToken);

        storedToken.RevokedAt.Should().NotBeNull();
        _authRepo.Verify(r => r.AddRefreshTokenAsync(It.Is<RefreshToken>(t =>
            t.TokenHash == newTokenHash && t.UserId == _testUser.Id
        ), default), Times.Once);
        _authRepo.Verify(r => r.SaveChangesAsync(default), Times.Once);
    }
}
```
